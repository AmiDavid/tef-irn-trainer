const { chromium } = require("playwright");
const { spawn } = require("node:child_process");
const assert = require("node:assert").strict;

const server=spawn("python3",["-m","http.server","4173","--bind","127.0.0.1"],{stdio:"ignore"});
const base="http://127.0.0.1:4173/?native=qa";

async function waitServer(){
  for(let i=0;i<50;i++){
    try{const r=await fetch(base);if(r.ok)return}catch{}
    await new Promise(r=>setTimeout(r,100));
  }
  throw new Error("test server did not start");
}
async function visible(page,selector){
  const el=page.locator(selector);
  await el.waitFor({state:"visible"});
  return el;
}

(async()=>{
  await waitServer();
  const browser=await chromium.launch({headless:true,executablePath:process.env.PW_CHROMIUM||undefined});
  try{
    for(const vp of [{width:360,height:800},{width:412,height:915},{width:480,height:960}]){
      const page=await browser.newPage({viewport:vp});
      const errors=[];
      page.on("pageerror",e=>errors.push(e.message));
      await page.goto(base,{waitUntil:"networkidle"});
      const overflow=await page.evaluate(()=>document.documentElement.scrollWidth-window.innerWidth);
      assert.ok(overflow<=1,"horizontal overflow at "+vp.width+": "+overflow);
      const brand=await page.locator(".brand").boundingBox();
      const actions=await page.locator(".top-actions").boundingBox();
      assert.ok(brand && actions);
      assert.ok(brand.x+brand.width<=actions.x+1,"header collision at "+vp.width);
      assert.deepEqual(errors,[],"page errors at "+vp.width+": "+errors.join(" | "));
      await page.close();
    }

    const page=await browser.newPage({viewport:{width:412,height:915}});
    const pageErrors=[];
    page.on("pageerror",e=>pageErrors.push(e.message));
    await page.goto(base,{waitUntil:"networkidle"});

    await page.locator('#bottomNav [data-view="flashcards"]').click();
    assert.equal((await page.locator("h1").filter({hasText:"Vocabulaire & expressions"}).count()),1);
    assert.equal(await page.locator("[data-flash-answer]").count(),4);
    assert.equal((await page.locator("#flashFeedback").textContent()).trim(),"");
    await page.locator("[data-flash-answer]").first().click();
    assert.ok((await page.locator("#flashFeedback").textContent()).trim().length>0);

    await page.locator('[data-vocab-tab="list"]').click();
    assert.equal(await page.locator(".vocab-summary .vocab-stat").count(),4);
    assert.ok(await page.locator(".vocab-item").count()>=15);
    assert.ok((await page.locator('[data-vstatus="good"]').textContent()).includes("Bien connu"));
    assert.ok((await page.locator('[data-vstatus="bad"]').textContent()).includes("À apprendre"));

    const targets=[
      ["plan","Plan & Gantt"],
      ["grammar","Grammaire B2"],
      ["errors","Mes erreurs récurrentes"],
      ["scan","Scanner mes notes de français"],
      ["mock","Examens blancs"],
      ["official","TEF officiel"]
    ];
    for(const [id,label] of targets){
      await page.locator("#mobileMoreBtn").click();
      await visible(page,'[data-more-view="'+id+'"]');
      await page.locator('[data-more-view="'+id+'"]').click();
      assert.ok((await page.locator("#view-"+id+" h1").textContent()).includes(label));
    }

    await page.locator("#mobileMoreBtn").click();
    await page.locator('[data-more-view="grammar"]').click();
    const grammarCount=await page.locator(".grammar-item").count();
    assert.ok(grammarCount>=28,"all grammar topics listed");
    /* Every topic opens its own page with rule, examples, notes, summary and a practice set. */
    for(let i=0;i<grammarCount;i++){
      const id=await page.locator(".grammar-item").nth(i).getAttribute("data-g");
      await page.evaluate(id=>{location.hash="#g/"+id},id);
      await visible(page,"#view-gtopic.active .topic-hero h1");
      for(const h of ["La règle","Exemples","Mes notes","À retenir"]) assert.ok((await page.locator("#gtopicBody").textContent()).includes(h),id+" missing section "+h);
      assert.ok(await page.locator("#gtopicBody .ex").count()>=3,id+" examples");
    }
    await page.locator("#mobileMoreBtn").click();
    await page.locator('[data-more-view="grammar"]').click();
    await page.locator('.grammar-item[data-g="g7"]').click();
    await visible(page,"#view-gtopic.active");
    await page.locator("#topicNotes").fill("Les si n’aiment pas les -rais.");
    await page.waitForTimeout(700);
    /* Practice page: answer each exercise; typed answers use the expected answer, then one deliberate mistake. */
    await page.locator("[data-practice]").first().click();
    await visible(page,"#view-gpractice.active .exercise");
    const items=await page.evaluate(()=>window.TEF_TOPIC.practiceItems("g7"));
    assert.ok(items.length>=12);
    for(let i=0;i<items.length;i++){
      const x=items[i];
      if(x.type==="mcq"){await page.locator("#gpracticeBody [data-ok='1']").click()}
      else if(x.type==="order"){
        const target=x.answers[0].replace(/[.!?…]+$/,"").split(" ");
        const used=new Set();
        for(const w of target){
          const toks=page.locator("#orderBank .token");const n=await toks.count();
          for(let k=0;k<n;k++){const t=toks.nth(k);const idx=await t.getAttribute("data-t");if(used.has(idx))continue;if((await t.textContent()).toLocaleLowerCase("fr")===w.toLocaleLowerCase("fr")){await t.click();used.add(idx);break}}
        }
        await page.locator("#exCheck").click();
      }else{
        const ans=i===2?"mauvaise réponse":x.answers[0];
        await page.locator("#exInput").fill(ans);
        await page.locator("#exCheck").click();
      }
      const fb=await page.locator("#exFb").textContent();
      if(i===2&&x.type!=="mcq"&&x.type!=="order")assert.ok(fb.includes("Pas tout à fait"),"wrong typed answer must be rejected");
      else assert.ok(fb.includes("Correct"),"exercise "+i+" ("+x.type+") should accept the expected answer: "+fb);
      await page.locator("#exNext").click();
    }
    await visible(page,"#redoAll");
    assert.ok(await page.evaluate(()=>window.TEF_APP.S.gnotes.g7.includes("rais")),"notes saved");
    /* Accent-only differences are flagged as « presque ». */
    assert.equal(await page.evaluate(()=>window.TEF_TOPIC.check("Je préfère",["Je préfère."]).verdict),"ok");
    assert.equal(await page.evaluate(()=>window.TEF_TOPIC.check("je prefere",["Je préfère."]).verdict),"accents");
    assert.equal(await page.evaluate(()=>window.TEF_TOPIC.check("J'ai  fini !",["J’ai fini."]).verdict),"ok");
    await page.locator("#mobileMoreBtn").click();
    await page.locator('[data-more-view="grammar"]').click();

    /* Grammar drills record progress; conjugation tables render. */
    await page.locator("#conjInput").fill("s’inscrire");
    await page.locator("#conjGo").click();
    assert.ok((await page.locator("#conjOut").textContent()).includes("elle s’est inscrite"),"conjugation table missing être agreement");
    assert.equal(await page.locator("#conjOut .conj-table").count(),9);

    /* Vocabulary search ranks whole-word matches first and filters by theme / verb group. */
    await page.locator('#bottomNav [data-view="flashcards"]').click();
    await page.locator('[data-vocab-tab="list"]').click();
    await page.locator("#vocabSearch").fill("logement");
    assert.equal(await page.locator(".vocab-item strong").first().textContent(),"le logement");
    await page.locator("#vocabSearch").fill("");
    await page.locator("#vocabPos").selectOption("verbe:2");
    const g2=await page.locator(".vocab-detail").allTextContents();
    assert.ok(g2.length>0&&g2.every(t=>t.includes("2e groupe")),"verb-group filter");
    await page.locator("#vocabPos").selectOption("all");

    /* Gender & plural flashcards. */
    await page.locator('[data-vocab-tab="practice"]').click();
    await page.locator('[data-deck-mode="gender"]').click();
    assert.ok((await page.locator("#flashcard .front").textContent()).includes("Genre et pluriel"));
    await page.locator("[data-flash-answer]").first().click();
    await page.locator('[data-rate="good"]').click();
    await page.locator('[data-deck-mode="mix"]').click();

    /* Reading practice series: immediate feedback with explanation. */
    await page.locator('#bottomNav [data-view="practice"]').click();
    await page.locator('#practiceTabs [data-tab="reading"]').click();
    await page.locator("#startSeries").click();
    await visible(page,"#examRunner .doc-box");
    for(let i=0;i<5;i++){
      await page.locator("#examRunner [data-ans]").first().click();
      assert.ok((await page.locator("#rFb").textContent()).trim().length>10,"practice feedback");
      await page.locator("#rNext").click();
    }
    await visible(page,"#rAgain");
    await page.locator("#rClose").click();

    /* Mini mock exam: reading (6 q) then listening (6 q, audio once), saved to history. */
    await page.locator('#bottomNav [data-view="mock"]').click();
    await page.locator('[data-mock="mini"]').click();
    await page.locator("#rGo").click();
    assert.match(await page.locator("#rTimer").textContent(),/^0?\d:\d\d$/);
    for(let i=0;i<6;i++){await page.locator("#examRunner [data-ans]").nth(i%4).click();await page.locator("#rNext").click()}
    await page.locator("#rGo").click();
    page.on("dialog",d=>d.accept());
    for(let i=0;i<6;i++){
      await page.locator("#rPlay").click();
      assert.ok(await page.locator("#rPlay").isDisabled(),"audio must be playable only once");
      assert.equal(await page.locator("#rPrev").count(),0,"no going back in listening");
      await page.locator("#examRunner [data-ans]").first().click();
      await page.locator("#rNext").click();
    }
    await page.locator(".result-card").first().waitFor({state:"visible"});
    assert.equal(await page.locator(".result-card").count(),2);
    await page.locator("#rClose").click();
    assert.equal(await page.locator("#mockHistory tbody tr").count(),1);

    /* Dashboard levels move once enough comprehension answers exist. */
    await page.locator('#bottomNav [data-view="today"]').click();
    const notes=await page.locator("#skillList small").allTextContents();
    assert.ok(notes.some(t=>/questions récentes/.test(t)),"reading/listening estimate should come from practice data");

    assert.deepEqual(pageErrors,[],"runtime errors: "+pageErrors.join(" | "));
    await page.close();
    console.log("UI QA OK");
  } finally {
    await browser.close();
    server.kill("SIGTERM");
  }
})().catch(err=>{server.kill("SIGTERM");console.error(err);process.exit(1)});
