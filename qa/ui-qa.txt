const base="http://127.0.0.1:4173/?native=qa";

async function waitServer(){
  for(let i=0;i<50;i++){
    try{const r=await fetch(base);if(r.ok)return}catch{}
    await new Promise(r=>setTimeout(r,100));
  }
  throw new Error("test server did not start");
}
async function visible(page,selector){
  const el=page.locator(selector);
  await el.waitFor({state:"visible"});
  return el;
}

    await page.evaluate(()=>{delete TEF_APP.S.nbMode;if(TEF_APP.S.copied)delete TEF_APP.S.copied.g30;TEF_TOPIC.copy("g30")});
    await visible(page,"#view-gcopy.active #copyMark");
    await page.locator("#copyMark").click();
    await visible(page,"#view-gcopy.active .nb-fiche");
    assert.equal(await page.evaluate(()=>TEF_APP.S.copied.g30.mode),"fiche");
    assert.equal(await page.locator("#copyMark").count(),0,"no button once the page is recorded");
    await page.evaluate(()=>{delete TEF_APP.S.copied.g30;TEF_APP.save()});
    /* A lesson without a fiche falls back to the short page: « À retenir » first, the detailed rules folded. */
    await page.evaluate(()=>{window.__fiche=TEF_APP.D.grammarFiche.g11;delete TEF_APP.D.grammarFiche.g11;TEF_APP.S.nbMode="fiche";TEF_TOPIC.copy("g11")});
    await visible(page,'#view-gcopy.active [data-nbmode="fiche"][aria-pressed="true"]');
    assert.equal((await page.locator("#gcopyBody .notebook h2").first().textContent()).trim(),"À retenir","short page starts with « À retenir »");
    assert.equal(await page.locator("#gcopyBody details.nb-fold:not([open]) .nb-rule").count(),1,"detailed rules folded");
    assert.equal(await page.locator("#gcopyBody table").count(),3,"tables kept in the short page");
    const shortCopy=await page.evaluate(()=>{TEF_TOPIC.markCopied("g11");const c=TEF_APP.S.copied.g11;return {mode:c.mode,n:c.keys.length,want:TEF_TOPIC.notebookLines("g11","short").length,full:TEF_TOPIC.notebookLines("g11","full").length,upd:TEF_TOPIC.updatesOf("g11").length}});
    assert.equal(shortCopy.mode,"short");assert.equal(shortCopy.n,shortCopy.want);assert.ok(shortCopy.full>shortCopy.want);
    assert.equal(shortCopy.upd,0,"rules left out of a short copy are never asked for");
    await page.evaluate(()=>{TEF_APP.D.grammarFiche.g11=window.__fiche;delete TEF_APP.S.nbMode;delete TEF_APP.S.copied.g11;delete TEF_APP.S.copied.g5;TEF_APP.save()});
    await page.evaluate(()=>{delete window.__TEF_TODAY__;delete TEF_APP.S.course;TEF_APP.save();TEF_DAILY.render()});

    assert.deepEqual(pageErrors,[],"runtime errors: "+pageErrors.join(" | "));
    await page.close();
    console.log("UI QA OK");
  } finally {
    await browser.close();
    server.kill("SIGTERM");
  }
})().catch(err=>{server.kill("SIGTERM");console.error(err);process.exit(1)});
