const nonempty = v => typeof v === "string" && v.trim().length > 0;
const unique = arr => new Set(arr.map(x => String(x).trim().toLocaleLowerCase("fr"))).size === arr.length;
const safe = (text, label) => {
  const s=String(text||"");
  const bad=[
    /C’est l’examen auquel je me prépare demande/i,
    /\bbeaucoup des commerces\b/i,
    /\bde les habitants\b/i,
    /pour que vous prendre/i,
    /une nouvelle travaille/i,
    /prendre son retrait/i,
    /une course de cuisine/i,
    /un magazine du quartier/i,
    /\bun nouvelle travail\b/i,
    /\bundefined\b/i,
    /\bnull\b/i,
    /TODO/i
  ];
  for(const re of bad) assert.ok(!re.test(s),label+" contains suspicious learner-error text: "+s);
};

assert.ok(D,"TEF_DATA missing");

assert.equal(D.official.format.listening.parts,2);
assert.equal(D.official.format.listening.audioOnce,true);
assert.equal(D.official.format.writing.minutes,30);
assert.equal(D.official.format.writing.sections[0].id,"A");
assert.equal(D.official.format.writing.sections[0].minutes,10);

assert.ok(D.official.links.length>=4);

assert.ok(unique(D.errors.map(x=>x.id)));

assert.ok(unique(D.vocab.map(x=>x.id)));
assert.ok(unique(D.vocab.map(x=>x.front)));
for(const v of D.vocab){
  for(const k of ["front","back","example","tag"]) assert.ok(nonempty(v[k]),`vocab ${v.id} missing ${k}`);
  safe(v.front,`vocab ${v.id} front`);safe(v.example,`vocab ${v.id} example`);

assert.ok(unique(D.grammar.map(x=>x.id)));
const allowedStatuses=new Set(["weak","learning","untested","reliable","mastered"]);
for(const g of D.grammar){
  assert.ok(nonempty(g.title));assert.ok(nonempty(g.short));assert.ok(nonempty(g.details));
  assert.ok(allowedStatuses.has(g.status),"bad grammar status "+g.id);
  assert.ok(Array.isArray(g.examples)&&g.examples.length>=2,"not enough examples "+g.id);

  [g.short,g.details,...g.examples,g.drill.prompt,g.drill.choices[g.drill.answer]].forEach((x,i)=>safe(x,`grammar ${g.id} text ${i}`));

/* Prompts must follow the official TEF IRN task types (verified on lefrancaisdesaffaires.fr). */
const LEVELS=["A2","B1","B2","C1"];
for(const p of D.writingPrompts){
  assert.ok(["A","B"].includes(p.type),"writing type "+p.id);
  assert.deepEqual([p.minutes,p.minWords],p.type==="A"?[10,40]:[20,100],"writing constraints "+p.id);
  assert.ok(nonempty(p.prompt)&&nonempty(p.title),"writing prompt text "+p.id);
  safe(p.prompt,"writing "+p.id);
}
for(const p of D.speakingPrompts){
  assert.ok(["A","B"].includes(p.type),"speaking type "+p.id);
  assert.equal(p.minutes,5,"speaking minutes "+p.id);
  assert.ok(nonempty(p.prompt)&&nonempty(p.title)&&nonempty(p.examiner),"speaking prompt/examiner "+p.id);
  if(p.type==="A") assert.ok(Array.isArray(p.points)&&p.points.length>=4,"speaking A points "+p.id);
  if(p.type==="B") assert.match(p.prompt,/décision|décider/i,"speaking B must be « aider un(e) ami(e) à prendre une décision »: "+p.id);
  safe(p.prompt,"speaking "+p.id);
}
assert.ok(unique([...D.writingPrompts,...D.speakingPrompts].map(p=>p.id)),"prompt IDs duplicate");
for(const t of ["A","B"]){
  assert.ok(D.writingPrompts.filter(p=>p.type===t).length>=10,"need ≥10 writing prompts "+t);
  assert.ok(D.speakingPrompts.filter(p=>p.type===t).length>=10,"need ≥10 speaking prompts "+t);
}

/* Banks: comprehension items carry a CEFR level and an explanation; there must be enough per level for a full mock. */
for(const kind of ["reading","listening"]){
  for(const q of D[kind]){
    assert.ok(LEVELS.includes(q.level),kind+" "+q.id+" bad level "+q.level);
    assert.ok(nonempty(q.explain),kind+" "+q.id+" missing explanation");
    safe(q.explain,kind+" "+q.id+" explain");
    if(q.lines){
      assert.ok(q.lines.every(l=>Array.isArray(l)&&/^(F|H)2?$/.test(l[0])&&nonempty(l[1])),kind+" "+q.id+" bad dialogue lines");
      assert.equal(q.text,q.lines.map(l=>"— "+l[1]).join("\n"),kind+" "+q.id+" text must mirror dialogue lines");
    }
  }
  const need={A2:3,B1:6,B2:8,C1:3};
  for(const L of LEVELS) assert.ok(D[kind].filter(q=>q.level===L).length>=need[L]*2,kind+": not enough "+L+" items for two different full mocks");
}

/* Bank typography: curated bank text uses the typographic apostrophe. */
const bankStrings=[];
const walk=v=>{if(typeof v==="string")bankStrings.push(v);else if(Array.isArray(v))v.forEach(walk);else if(v&&typeof v==="object")Object.values(v).forEach(walk)};
walk(BANK);
for(const s of bankStrings) assert.ok(!s.includes("'"),"ASCII apostrophe in bank text: "+s.slice(0,80));

/* Vocabulary metadata. */
const POS=new Set(["nom","verbe","adj","adv","expr","connecteur"]);
for(const v of BANK.vocab){
  assert.ok(POS.has(v.pos),"vocab pos "+v.id);
  if(v.pos==="nom"){assert.ok(["m","f"].includes(v.gender),"noun gender "+v.id);assert.match(v.front,/^(le |la |l’|les )/,"noun article "+v.id);assert.match(v.plural||"",/^les /,"noun plural "+v.id)}
  if(v.pos==="adj")assert.ok(nonempty(v.fem),"adjective feminine "+v.id);
  if(v.pos==="verbe"){
    assert.ok([1,2,3].includes(v.group),"verb group "+v.id);assert.ok(["avoir","être"].includes(v.aux),"verb aux "+v.id);
    const c=CONJ.conjugate(v.front);
    assert.ok(c,"conjugation engine does not know "+v.front);
    assert.equal(c.aux,v.aux,"aux mismatch vocab vs engine: "+v.front);
    assert.equal(c.group,v.group,"group mismatch vocab vs engine: "+v.front);
  }
}
assert.ok(unique(D.vocab.map(x=>x.front)),"vocab fronts duplicate across banks");

/* Grammar: every card is placed in the learning path; every drill is valid. */
for(const g of D.grammar){
  assert.ok(nonempty(g.section)&&Number.isInteger(g.order),"grammar path "+g.id);
  for(const d of [g.drill,...(g.drills||[])]){
    assert.ok(Array.isArray(d.choices)&&d.choices.length>=3&&unique(d.choices),"grammar drill choices "+g.id);
    assert.ok(Number.isInteger(d.answer)&&d.answer>=0&&d.answer<d.choices.length,"grammar drill answer "+g.id);
    safe(d.choices[d.answer],"grammar "+g.id+" drill answer");safe(d.prompt,"grammar "+g.id+" drill prompt");
  }
  for(const m of g.mistakes||[]){assert.ok(nonempty(m.wrong)&&nonempty(m.right)&&nonempty(m.why),"grammar mistake "+g.id);assert.notEqual(m.wrong,m.right);safe(m.right,"grammar "+g.id+" mistake fix")}
}
assert.ok(unique(D.grammar.map(g=>g.order)),"grammar order collision");

  ...D.listening.flatMap(q=>[q.text,q.question,q.choices[q.answer]])
].join(" ");
