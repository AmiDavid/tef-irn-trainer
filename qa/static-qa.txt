const fs = require("node:fs");
const assert = require("node:assert").strict;

const read = p => fs.readFileSync(p,"utf8");
const html=read("index.html");
const css=read("styles.css");
const app=read("app.js");
const manifest=JSON.parse(read("manifest.webmanifest"));
const android=read("android/app/src/main/java/io/github/amidavid/tefb2coach/MainActivity.java");
const gradle=read("android/app/build.gradle.kts");

for (const f of ["index.html","styles.css","data.js","app.js","connected.js","manifest.webmanifest","sw.js"]) assert.ok(fs.existsSync(f), "missing "+f);
assert.ok(html.includes("Vocabulaire & expressions"));
assert.ok(html.includes('data-vocab-tab="list"'));
assert.ok(html.includes('id="mobileMoreBtn"'));
assert.ok(html.includes("v1.2 • build 8"));
assert.ok(html.includes("app.js?v=8"));
assert.ok(css.includes("@media(max-width:680px)"));
assert.ok(css.includes(".vocab-summary"));
assert.ok(app.includes("clearNativeCaches"));
assert.equal((app.match(/(?<!\$)\$\([^)]*\)\.forEach/g)||[]).length,0);
assert.ok(android.includes("WindowInsetsCompat.Type.systemBars()"));
assert.ok(android.includes("?native=android&v=8"));
assert.ok(gradle.includes('versionName = "1.2"'));
assert.equal(manifest.display,"standalone");
assert.ok(manifest.icons.some(x=>x.sizes==="192x192" && x.type==="image/png"));
assert.ok(manifest.icons.some(x=>x.sizes==="512x512" && x.type==="image/png"));

console.log("STATIC QA OK");
